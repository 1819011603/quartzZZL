    exts = ["jpg", "jpeg", "png"]
    from multiprocessing import Pool
    params = []
    for picPath in getCwdFilesPath(os.getcwd() + os.sep):
        ext = getFileExt(picPath)
        if not (ext in exts):
            continue
        params.append((picPath, 50, ext))
    s = time.time()
    with Pool() as pool:
        pool.starmap(compress_image, params)
