```go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io/ioutil"
	"net/http"
)

type Data struct {
	LongUrl        string `json:"LongUrl"`
	TermOfValidity string `json:"TermOfValidity"`
}

type ResponseData struct {
	ShortUrls []map[string]string
}

func getBaiduShortUrl(u string) {
	resp, err := http.Get(u)
	if err != nil {
		panic(err)
	}
	url := resp.Request.URL.String()

	data := []Data{
		Data{
			LongUrl:        url,
			TermOfValidity: "long-term",
		},
	}

	requestBody, err := json.Marshal(data)
	if err != nil {
		panic(err)
	}
	req, err := http.NewRequest("POST", "https://dwz.cn/api/v3/short-urls", bytes.NewBuffer(requestBody))
	if err != nil {
		panic(err)
	}
	req.Header.Set("Content-Type", "application/json;charset=UTF-8")
	req.Header.Set("Dwz-Token", "27da6d5a6991149015e23cd90e695920")

	client := &http.Client{}
	resp, err = client.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()

	responseBytes, err := ioutil.ReadAll(resp.Body)
	if err != nil {
		panic(err)
	}

	responseData := ResponseData{}
	err = json.Unmarshal(responseBytes, &responseData)
	if err != nil {
		panic(err)
	}

	fmt.Println("转化后的百度短链为: " + responseData.ShortUrls[0]["ShortUrl"])
}

var (
	url string
)

func main() {
	for {
		fmt.Print("输入需要转化的短链或长链: ")
		fmt.Scanln(&url)
		getBaiduShortUrl(url)
		fmt.Println()
	}
}
```
