import os

coreThreads = ThreadPoolExecutor(10)

input_text = sg.Multiline(default_text="", key='translate-input', size=(100, 5), enable_events=True)

# Define the layout of the GUI
layout = [
    [sg.Text("Enter some text:"),
     input_text],
    [sg.Button('翻译', bind_return_key=True), sg.Button('时间戳'), sg.Button("json格式化"), sg.Button("解析日志json")],

def english_to_datetime(time_str):
    from datetime import datetime, timezone, timedelta
    # 将字符串转换为datetime对象（东八区时间）
    dt_object = datetime.strptime(time_str, "%b %d, %Y %I:%M:%S %p")
    dt_object = dt_object.replace(tzinfo=timezone(timedelta(hours=8)))  # 设置时区为东八区
    # 将datetime对象转换为时间戳（以秒为单位）
    timestamp = int(dt_object.timestamp())
    return timestamp


def timestamp_to_datetime(timestamp, milliseconds=False, timezone="Asia/Shanghai"):
    if milliseconds:
        timestamp /= 1000  # Convert milliseconds to seconds if needed
    utc_datetime = datetime.utcfromtimestamp(timestamp)
    utc_datetime = utc_datetime.replace(tzinfo=pytz.utc)  # Set the timezone to UTC

    # Convert to the desired timezone
    local_timezone = pytz.timezone(timezone)
    local_datetime = utc_datetime.astimezone(local_timezone)

    # Format the datetime object as a string

    formatted_datetime = local_datetime.strftime('%Y-%m-%d %H:%M:%S.%f')[:-3]

    return formatted_datetime


def parse_log_json(input_text):
    """
    解析日志json
    :return:
    """
    input_text = json.loads(input_text, strict=False)
    messages = str(input_text.get("messages"))
    l = messages.find("{")
    r = messages.rfind("}")
    content = None
    flag = False
    while l < r:
        tmp_r = r
        while l < tmp_r:
            try:
                content = json.loads(messages[l:tmp_r + 1], strict=False)
                flag = True
                break
            except Exception:
                pass
            tmp_r = messages[:tmp_r].rfind("}")
        if flag:
            break
        l = messages[l + 1:r + 1].find("{")
        r = messages[l:r].rfind("}")
    if r == -1:
        return "解析异常"
    pattern = r'"(\w+)": *(["\']\w{3} \d{1,2}, \d{4} \d{1,2}:\d{2}:\d{2} [APMapm]{2}["\'])'
    content1 = json.dumps(content)
    times = re.findall(pattern, content1)
    if times:
        for t in times:
            content[t[0]] = str(english_to_datetime(eval(t[1])))
    return content


def datetime_string_to_timestamp(datetime_str, milliseconds=False):
    dt_format = '%Y-%m-%d %H:%M:%S.%f' if '.' in datetime_str else '%Y-%m-%d %H:%M:%S'
    dt_object = datetime.strptime(datetime_str, dt_format)

    if milliseconds:
        timestamp = int(dt_object.timestamp() * 1000)  # Convert to milliseconds
    else:
        timestamp = int(dt_object.timestamp())

    return str(timestamp)


digits_pattern = r"^[0-9]+$"
time_pattern = r"^[0-9]{4}-[0-9]{2}-[0-9]{2} [0-9]{2}:[0-9]{2}:[0-9]{2}(\.[0-9]{3})*"


# Define an asynchronous function to simulate a time-consuming task

    return translate__.translate(input_text)


def select_all(element):
    # 选择Combobox输入框中的所有文本

def result_add(*params):
    params = [str(i) for i in list(params)]
    if len(params) > 1:
        window["translate-output"].update(value=os.linesep.join(params))
    else:
        window["translate-output"].update(value=params[0])
    window.Refresh()


def deal解析日志json():
    try:
        res = json.dumps(parse_log_json(pyperclip.paste()), ensure_ascii=False, sort_keys=True).replace("'",
            '"').replace(
            "True", "true").replace("False", "false")
        print(res)
        result_add(str(res))
    except Exception as e:
        result_add(str(e))


translate__ = TranslateSelf()

    text = window["result"].get().strip(os.linesep)
    text = text.strip()

        res = res[1:]

window = sg.Window('Async Example', layout, enable_close_attempted_event=True, finalize=True)  # 必须设置 finalize

while True:

        if re.search(digits_pattern, input_text) is not None:

                timestamp_to_datetime(time_sec, len(input_text) != len(cur_time))[:-4],

            coreThreads.submit(lambda: result_add(
                datetime_string_to_timestamp(input_text, False),
                datetime_string_to_timestamp(input_text, True)

        else:
            coreThreads.submit(lambda: result_add(
                english_to_datetime(input_text),
                english_to_datetime(input_text) * 1000
            ))

        input_text = values['translate-input']

    elif event == '解析日志json':

                input_text_commands_mapping.get(str(event)[len('translate-input'):])(
                                   window['translate-input'])
            )
