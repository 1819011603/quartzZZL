**起因：** 写`Chrome`扩展的时候，`input` 框自动填写内容后，无法触发表单的提交，需要出发[input事件](https://so.csdn.net/so/search?q=input%E4%BA%8B%E4%BB%B6&spm=1001.2101.3001.7020)。


https://blog.csdn.net/github_39570717/article/details/130129215

```javascript
// ==UserScript==
// @name         快速点击登录
// @namespace    http://tampermonkey.net/
// @version      0.1
// @description  快速点击登录
// @author       张泽灵
// @grant        none
// @match        https://test-cas.baijia.com/cas/*
// @match        https://cas.baijia.com/cas/*
// @downloadURL https://update.greasyfork.org/scripts/393020/%E5%BF%AB%E9%80%9F%E5%A4%8D%E5%88%B6cookie.user.js
// @updateURL https://update.greasyfork.org/scripts/393020/%E5%BF%AB%E9%80%9F%E5%A4%8D%E5%88%B6cookie.meta.js
// ==/UserScript==

(function() {
    'use strict';

    function setValue(input11, value){
        // 创建一个新的 input 事件
        const event2 = new Event('input', {
            bubbles: true, // 事件是否冒泡
            cancelable: true // 事件是否可以取消
        });
        let lastValue = input11.value;

            input11.value = value;
	//  React15
	event2.simulated = true;
	//  React16 内部定义了descriptor拦截value，此处重置状态
	let tracker = input11._valueTracker;
	if (tracker) {
		tracker.setValue(lastValue);
	}
        // 确保输入框存在
        if (input11) {
            // 创建一个新的 input 事件
            input11.dispatchEvent(event2);
            console.info('输入框已发布事件');

        } else {
            console.error('输入框未找到');
        }
    }
    // Function to click the button
    function clickButton() {



        // Use document.evaluate to find the button with XPath
        // 定义XPath表达式
        var xpath1 = '/html/body/div/div/div/div/div[2]/div[1]/div/div/span/span/input'
        const result1 = document.evaluate(xpath1, document, null, XPathResult.FIRST_ORDERED_NODE_TYPE, null);
        const input1 = result1.singleNodeValue;
        setValue(input1, "zhangzeling");

        var xpath2 = '/html/body/div/div/div/div/div[2]/div[2]/div/div/span/span/input'
        const result2 = document.evaluate(xpath2, document, null, XPathResult.FIRST_ORDERED_NODE_TYPE, null);
        const input2 = result2.singleNodeValue;
        setValue(input2, "密码");


        const xpath = '/html/body/div/div/div/div/div[2]/button';
        const result = document.evaluate(xpath, document, null, XPathResult.FIRST_ORDERED_NODE_TYPE, null);
        const button = result.singleNodeValue;
        if (button) {
            button.focus()
            // 触发事件
            button.click();
            console.log('Button clicked!');
        } else {
            console.log('Button not found.');
        }
    }
         try {
              clickButton();
        } catch (error) {
            console.error("发生错误:", error.message); // 捕获并处理异常
        }
    // DOM 加载完成后执行的代码
    console.log("DOM 完全加载！");

    // Wait for the page to fully load
    setInterval(function() {
        try {
              clickButton();
        } catch (error) {
            console.error("发生错误:", error.message); // 捕获并处理异常
        }
    }, 1000);
})();
```
