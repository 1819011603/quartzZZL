2. 对结果出现的依赖排除 log4j-to-slf4j即可

```xml
           <exclusions>
                <exclusion>
                    <groupId>org.apache.logging.log4j</groupId>
                    <artifactId>log4j-to-slf4j</artifactId>
                </exclusion>
            </exclusions>
```

### 日志级别根据apollo自动刷新 

logging.level.com.example=DEBUG
``` java
import com.ctrip.framework.apollo.Config;
import com.ctrip.framework.apollo.ConfigService;
import com.ctrip.framework.apollo.model.ConfigChangeEvent;
import com.ctrip.framework.apollo.model.ConfigChange;
import org.apache.logging.log4j.Level;
import org.apache.logging.log4j.core.config.Configurator;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.event.EventListener;
import org.springframework.stereotype.Component;

@Component
public class LogLevelConfigListener {

    // Apollo 配置中心的日志级别配置
    @Value("${logging.level.root:INFO}")
    private String logLevel;

    // 创建 Apollo Config 实例
    private Config config;

    public LogLevelConfigListener() {
        // 从 Apollo 配置中心加载配置
        config = ConfigService.getAppConfig();
        // 监听配置的变更
        config.addChangeListener(this::onConfigChange);
    }

    /**
     * 配置变更的回调函数，处理日志级别变更
     *
     * @param changeEvent Apollo 配置变更事件
     */
    private void onConfigChange(ConfigChangeEvent changeEvent) {
        // 如果日志级别发生变化，调整日志级别
        for (String changedKey : changeEvent.changedKeys()) {
            ConfigChange change = changeEvent.getChange(changedKey);
            // 这里判断是否是日志级别配置的变更
            if ("logging.level.root".equals(changedKey)) {
                String newLogLevel = change.getNewValue();
                updateLogLevel(newLogLevel);
            }
        }
    }

    /**
     * 根据日志级别更新 Log4j2 配置
     *
     * @param newLogLevel 新的日志级别
     */
    private void updateLogLevel(String newLogLevel) {
        switch (newLogLevel) {
            case "DEBUG":
                Configurator.setLevel("com.yourpackage", Level.DEBUG);
                break;
            case "INFO":
                Configurator.setLevel("com.yourpackage", Level.INFO);
                break;
            case "WARN":
                Configurator.setLevel("com.yourpackage", Level.WARN);
                break;
            case "ERROR":
                Configurator.setLevel("com.yourpackage", Level.ERROR);
                break;
            default:
                Configurator.setLevel("com.yourpackage", Level.INFO);
        }
    }

    /**
     * 启动时初始化当前日志级别
     */
    @EventListener
    public void init() {
        updateLogLevel(this.logLevel);
    }
}
```
