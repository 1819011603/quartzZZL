def get_attr(*, employee: Employee, attr: EmployeeAttribute):
    """获取员工对象的指定属性值"""
    match attr:
        case EmployeeAttribute.ID:
            return employee.id
        case EmployeeAttribute.NAME:
            return employee.name
        case EmployeeAttribute.EMAIL:
            return employee.email
        case EmployeeAttribute.DISPLAY_NAME:
            return employee.display_name
        case EmployeeAttribute.AVATAR:
            return employee.avatar


def build_employee_from_mapping(mapping: Mapping[str, Any], force_check: bool = False) -> Employee:
    """从映射创建员工对象，支持查询完整信息"""
    # 如果数据完整，直接创建对象
    if _has_complete_employee_data(mapping) and not force_check:
        return Employee.model_validate(mapping)
    
    # 如果只有部分字段，查询完整信息
    from services.employee_service import EmployeeQueryService
    
    complete_data = EmployeeQueryService.batch_query_employees([dict(mapping)])
    if complete_data:
        mapping = complete_data[0]
    
    return Employee.model_validate(mapping)


def build_employees_from_mappings(mappings: Sequence[Mapping[str, Any]]) -> list[Employee]:
    """从映射列表创建员工对象列表，支持批量查询完整信息"""
    if not mappings:
        return []
    
    # 分离完整数据和不完整数据
    complete_employees = []
    incomplete_mappings = []
    
    for mapping in mappings:
        if _has_complete_employee_data(mapping):
            complete_employees.append(Employee.model_validate(mapping))
        else:
            incomplete_mappings.append(mapping)
    
    # 只对不完整的数据进行批量查询
    if incomplete_mappings:
        from services.employee_service import EmployeeQueryService
        
        complete_data_list = EmployeeQueryService.batch_query_employees(
            [dict(m) for m in incomplete_mappings]
        )
        complete_employees.extend([Employee.model_validate(data) for data in complete_data_list])
    
    return complete_employees


def _has_complete_employee_data(mapping: Mapping[str, Any]) -> bool:
    """检查员工数据是否完整，包含字段"""
    key_fields = {"id", "name", "email", "display_name", "avatar"}
    return all(field in mapping and mapping[field] for field in key_fields)

    @classmethod
    def batch_query_employees(cls, employee_data_list: list[dict[str, Any]]) -> list[dict[str, Any]]:
        """
        批量查询员工完整信息
        
        Args:
            employee_data_list: 员工数据列表，可能只包含部分字段
            
        Returns:
            完整的员工信息列表
        """
        if not employee_data_list:
            return []
        
        # 收集需要查询的关键字段
        ids_to_query = set()
        emails_to_query = set()
        names_to_query = set()
        
        for emp_data in employee_data_list:
            if emp_data.get("id"):
                ids_to_query.add(emp_data["id"])
            elif emp_data.get("email"):
                emails_to_query.add(emp_data["email"])
            elif emp_data.get("name"):
                names_to_query.add(emp_data["name"])
        
        # 批量查询员工信息
        employees_by_id = cls._query_employees_by_ids(list(ids_to_query))
        employees_by_email = cls._query_employees_by_emails(list(emails_to_query))
        employees_by_name = cls._query_employees_by_names(list(names_to_query))
        
        # 合并查询结果
        result = []
        for emp_data in employee_data_list:
            complete_employee = cls._merge_employee_data(
                emp_data, 
                employees_by_id, 
                employees_by_email, 
                employees_by_name
            )
            result.append(complete_employee)
        
        return result

    @classmethod

        """根据邮箱批量查询员工信息"""

    @classmethod
    def _merge_employee_data(
        cls, 
        original_data: dict[str, Any], 
        employees_by_id: dict[int, dict[str, Any]],
        employees_by_email: dict[str, dict[str, Any]],
        employees_by_name: dict[str, dict[str, Any]]
    ) -> dict[str, Any]:
        """合并员工数据，优先使用查询到的完整信息"""
        
        # 尝试通过不同方式查找完整信息
        complete_employee = None
        
        if original_data.get("id") and int(original_data["id"]) in employees_by_id:
            complete_employee = employees_by_id[int(original_data["id"])]
        elif original_data.get("email") and original_data["email"] in employees_by_email:
            complete_employee = employees_by_email[original_data["email"]]
        elif original_data.get("name") and original_data["name"] in employees_by_name:
            complete_employee = employees_by_name[original_data["name"]]
        
        if complete_employee:
            # 使用查询到的完整信息，但保留原始数据中的额外字段
            result = complete_employee.copy()
            # 只保留 JSON 可序列化的字段
            result.update({
                k: v for k, v in original_data.items()
                if k not in result and isinstance(v, (str, int, float, bool, type(None), dict, list))
            })
            return result
        else:
            raise ValueError("未找到员工信息:" + json.dumps(original_data))

        logger.error("headers is None, skip injecting billing and trace headers")
        return

    tid = get_trace_id()
    if tid:
        headers["X-B3-TraceId"] = str(tid)

    # 确定是否需要传递 run_user 和 nid
    should_pass_run_user = False
    should_pass_nid = False

    if billing_type == "run_user":

    elif billing_type == "nid":

            run_user_name = run_user.name

class EndUserService:
    """
    Service for managing end users.
    """

    @classmethod
    def get_or_create_end_user(cls, app_model: App, user_id: str | None = None) -> EndUser:
        """
        Get or create an end user for a given app.
        """

        return cls.get_or_create_end_user_by_type(InvokeFrom.SERVICE_API, app_model.tenant_id, app_model.id, user_id)

    @classmethod
    def get_or_create_end_user_by_type(
        cls, type: InvokeFrom, tenant_id: str, app_id: str, user_id: str | None = None
    ) -> EndUser:
        """
        Get or create an end user for a given app and type.
        """

        if not user_id:
            user_id = DefaultEndUserSessionID.DEFAULT_SESSION_ID

        with Session(db.engine, expire_on_commit=False) as session:
            end_user = (
                session.query(EndUser)
                .where(
                    EndUser.tenant_id == tenant_id,
                    EndUser.app_id == app_id,
                    EndUser.session_id == user_id,
                    EndUser.type == type,
                )
                .first()
            )

            if end_user is None:
                end_user = EndUser(
                    tenant_id=tenant_id,
                    app_id=app_id,
                    type=type,
                    is_anonymous=user_id == DefaultEndUserSessionID.DEFAULT_SESSION_ID,
                    session_id=user_id,
                    external_user_id=user_id,
                )
                session.add(end_user)
                session.commit()

        return end_user

    @classmethod
    def create_end_user_batch(
        cls, type: InvokeFrom, tenant_id: str, app_ids: list[str], user_id: str
    ) -> Mapping[str, EndUser]:
        """Create end users in batch.

        Creates end users in batch for the specified tenant and application IDs in O(1) time.

        This batch creation is necessary because trigger subscriptions can span multiple applications,
        and trigger events may be dispatched to multiple applications simultaneously.

        For each app_id in app_ids, check if an `EndUser` with the given
        `user_id` (as session_id/external_user_id) already exists for the
        tenant/app and type `type`. If it exists, return it; otherwise,
        create it. Operates with minimal DB I/O by querying and inserting in
        batches.

        Returns a mapping of `app_id -> EndUser`.
        """

        # Normalize user_id to default if empty
        if not user_id:
            user_id = DefaultEndUserSessionID.DEFAULT_SESSION_ID

        # Deduplicate app_ids while preserving input order
        seen: set[str] = set()
        unique_app_ids: list[str] = []
        for app_id in app_ids:
            if app_id not in seen:
                seen.add(app_id)
                unique_app_ids.append(app_id)

        # Result is a simple app_id -> EndUser mapping
        result: dict[str, EndUser] = {}
