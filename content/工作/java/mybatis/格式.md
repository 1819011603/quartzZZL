> 查询 根据条件查询
```xml
<select id="selectByUsernames" resultMap="BaseResultMap">
    <include refid="Base_Column_List"/>
    from urobot.ss_wp_work_wechat_base_info
    <if test="usernames != null and !usernames.isEmpty()">
        where username in
        <foreach collection="usernames" open="(" separator="," close=")" item="item">
            #{item,jdbcType=VARCHAR}
        </foreach>
    </if>
    <if test="usernames == null or usernames.isEmpty()">
        where 1=0
    </if>
</select>
```
