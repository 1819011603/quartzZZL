@Slf4j  
@Component  
public class BulkMsgSendTimer {  
    private static final String OK = "OK";  
    private static final String lockValue = "task";  
    ExecutorService threadPool = Executors.newFixedThreadPool(10);  
    private String msgLock = "msgLock";  
    /**  
     * 定时扫描数据的间隔时间  
     */  
    @Value("${scan.interval.time}")  
    private Integer scanTime;  
  
    @Value("${queue.expired.time}")  
    private Integer queueExpiredTime;  
  
    @Resource  
    private RobotWinServerRecordMapper recordMapper;  
  
    @Resource  
    private CasWxiduRecordPoMapper casWxiduRecordPoMapper;  
  
    @Resource  
    private RedisClient redisClient;  
  
    @Resource(name = "salesTransactionTemplate")  
    private TransactionTemplate transactionTemplate;  
  
    @Resource  
    private MultitaskConfig multitaskConfig;  
  
  
    @Scheduled(cron = "*/" + "${scan.interval.time}" + " * * * * ?")  
    public void scheduled() throws Exception {  
        // 只允许一个线程同时执行这一个方法  
        if (OK.equals(redisClient.set(msgLock, lockValue, "NX", "EX", 180))) {  
            try {  
                MDC.put(UNIQUE_ID, UUIDUtils.getUUID());  
                log.info("群发定时扫描任务开始执行");  
                RobotWinServerRecordPo robotWinServerRecordPo = new RobotWinServerRecordPo();  
                Date minTime = DateUtils.getDelayDate(new Date(), -20 * scanTime);  
                Date maxTime = DateUtils.getDelayDate(new Date(), scanTime);  
                robotWinServerRecordPo.setType(RobotWinServerDef.typeTiming);  
                // 1表示后端执行，前端的直接不管  
                robotWinServerRecordPo.setExecType(DefaultNumEnum.ONE.getCode());  
                // 查出需要排队的队列，以当前时间为基准计算前后x分钟，为防止程序执行的时间，应该比定时时间多那么一分钟  
                List<RobotWinServerRecordPo> robotWinServerRecordPos = recordMapper  
                        .selectRecords2Queue(robotWinServerRecordPo, minTime.getTime(), maxTime.getTime());  
                List<Integer> ids = robotWinServerRecordPos.stream().filter(po-> po.getSendTime() <= System.currentTimeMillis()).map((recordPo) -> recordPo.getId()).collect(Collectors.toList());  
                // 由定时发送转为排队中的群发任务ID  
                log.info("bulk send msg record timing to waiting id: {}", JSONObject.toJSONString(ids));  
                if (CollectionUtils.isNotEmpty(ids)) {  
                    // 这些任务直接进入排队中  
                    recordMapper.updateRecordsInQueue(RobotWinServerDef.typeWaiting, ids);  
                }  
                // 排队中的第一条消息消息准备进入队列  
                // 这个地方需要拿出所有排队中的，选出排队中的最新一条,排队的任务数据是否会很大？  
                // 配置一个过期时间，apoll配置  
                Date expiredDate = org.apache.commons.lang3.time.DateUtils.addHours(new Date(), -queueExpiredTime);  
                log.info("record expired Time {},{}", JSON.toJSONString(ids), expiredDate);  
                ArrayList<Integer> singleSendingTypes = Lists.newArrayList(RobotWinServerDef.typeWaiting);  
                List<RobotWinServerRecordPo> recordPos = recordMapper.selectRecordsInQueue(singleSendingTypes  
                        , expiredDate.getTime());  
                // 打印待发送的群发任务messageID  
                log.info("bulk send msg record type waiting message id: {}", JSONObject.toJSONString(recordPos.stream()  
                        .map(RobotWinServerRecordPo::getMessageId).collect(Collectors.toList())));  
                Map<String, List<RobotWinServerRecordPo>> recordsMap = recordPos.stream()  
                        .collect(Collectors.groupingBy(RobotWinServerRecordPo::getFromWxid));  
                // 避免分线程任务还没有执行完就释放了锁  
                CountDownLatch countDownLatch = new CountDownLatch(recordsMap.size());  
                recordsMap.keySet().forEach(  
                        fromWxid -> {  
                            // 在这个地方通过多个线程处理,只是拿map的数据,多线程完全ok  
                            threadPool.submit(new MsgSendTask(recordsMap, fromWxid, countDownLatch  
                                    , expiredDate.getTime(), transactionTemplate, multitaskConfig));  
                        }  
                );  
                countDownLatch.await();  
            } finally {  
                MDC.clear();  
                redisClient.del(msgLock);  
            }  
        }  
    }  
}  
  
  
@Slf4j  
class MsgSendTask implements Runnable {  
  
    Long expiredTime;  
    String fromWxid;  
    Map<String, List<RobotWinServerRecordPo>> recordsMap;  
    CountDownLatch countDownLatch;  
    TransactionTemplate runTransactionTemplate;  
    MultitaskConfig multitaskConfig;  
    private RobotWinServerRecordMapper recordMapper =  
            BeanContext.getApplicationContext().getBean(RobotWinServerRecordMapper.class);  
    private BulkMsgProducer msgProducer = BeanContext.getApplicationContext().getBean(BulkMsgProducer.class);  
    private static ScheduledThreadPoolExecutor scheduledThreadPoolExecutor = new ScheduledThreadPoolExecutor(10,new ThreadPoolExecutor.CallerRunsPolicy());  
    private static List<Integer> toRobotMqType = Arrays.asList(RobotWinServerDef.typeWaiting,RobotWinServerDef.typeTiming);  
  
    MsgSendTask(Map<String, List<RobotWinServerRecordPo>> recordsMap, String fromWxid  
            , CountDownLatch countDownLatch, Long expiredTime, TransactionTemplate transactionTemplate, MultitaskConfig multitaskConfig) {  
        this.recordsMap = recordsMap;  
        this.fromWxid = fromWxid;  
        this.expiredTime = expiredTime;  
        this.countDownLatch = countDownLatch;  
        this.runTransactionTemplate = transactionTemplate;  
        this.multitaskConfig = multitaskConfig;  
    }  
  
    @Override  
    public void run() {  
        try {  
            List<RobotWinServerRecordPo> recordPos = this.recordsMap.get(fromWxid);  
            // 需要判断当前fromWxid的队列中  
            Boolean multitaskResult = multitaskConfig.getMultitaskResult(fromWxid);  
            if (!multitaskResult){  
                // 串行  
                recordPos = SingleSendingCondition(recordPos);  
                if (recordPos == null) return;  
            }  
            List<RobotWinServerRecordPo> finalRecordPos = recordPos;  
            AtomicLong delay = new AtomicLong();  
            for (RobotWinServerRecordPo po : finalRecordPos) {  
                // 事务是保证mq发送失败 数据库可以回滚  
                long startDeliverTime = Math.max(10L, Math.min(po.getSendTime() - System.currentTimeMillis(), 60000L)) + delay.getAndIncrement() * 1200L;  
                Runnable sendRunnable = ()->{  
                    runTransactionTemplate.execute(transactionStatus -> {  
                        try {  
                            log.info("update record status is mq {}", po);  
                            // 需要更新消息的状态为队列中，这是使用消息队列之后的一个新的中间状态  
                            if (toRobotMqType.contains(po.getType()) && recordMapper.updateRecordTypeByMessageId(RobotWinServerDef.rocketMq, po.getMessageId()) > 0) {  
                                // 这个地方保证数据库更新状态在mq消费之前完成 别缩短时间!!! 慢就慢点  
                                SendResult sendResult = msgProducer.sendMsgRecord(po, 0L);  
                                log.info("timer wake up a record task {}, response {}"  
                                        , BizConf.gson.toJson(po), BizConf.gson.toJson(sendResult));  
                            }  
                            return true;  
                        } catch (Exception e) {  
                            transactionStatus.setRollbackOnly();  
                            log.error("scan update record status mq fail {}", JSON.toJSONString(po), e);  
                            return false;                        }  
                    });  
                };  
                scheduledThreadPoolExecutor.schedule(sendRunnable, startDeliverTime,TimeUnit.MILLISECONDS);  
            }  
        } finally {  
            log.info("close countDownLatch is starting");  
            countDownLatch.countDown();  
        }  
    }  
  
    private List<RobotWinServerRecordPo> SingleSendingCondition(List<RobotWinServerRecordPo> recordPos) {  
        List<Integer> types = Lists.newArrayList(RobotWinServerDef.rocketMq, RobotWinServerDef.typeSending,  
                RobotWinServerDef.typePause);  
        recordPos = recordPos.subList(0, 1);  
        List<RobotWinServerRecordPo> records = recordMapper.selectRecords4Wxid(fromWxid,  
                types, expiredTime);  
        if (CollectionUtils.isNotEmpty(records)) {  
            log.info("bugMsg has running task {},{}", JSON.toJSONString(recordPos), JSON.toJSONString(records));  
            // mq中已经有当前账号正在处理的任务  
            return null;  
        }  
        return recordPos;  
    }  
}
